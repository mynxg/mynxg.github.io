redis、naocs一键启动脚本

nacos-redis.bat

@echo off
echo Starting Redis...
start "" /D "E:\Redis-x64-5.0.14.1" .\redis-server.exe redis.windows.conf

echo Starting Nacos...
start "" /D "E:\nacos\bin" .\startup.cmd -m standalone

echo Both Redis and Nacos are started in separate windows.
pause

jar一键启动脚本

cloud-start.bat

@echo off
echo start  jar .....

:: 1、start gateway.jar
start "JAR 1 Window"  cmd /c java -Xms128m -Xmx128m  -Dfile.encoding=UTF-8  -jar "C:\Users\Administrator\Desktop\cloud-jar\yudao-gateway.jar"

:: 等待30秒
timeout /t 30 /nobreak >nul

::  2、start yudao-module-infra-biz.jar
start "JAR 2 Window"  cmd /c java -Xms128m -Xmx128m  -Dfile.encoding=UTF-8  -jar "C:\Users\Administrator\Desktop\cloud-jar\yudao-module-infra-biz.jar"

:: 等待30秒
timeout /t 30 /nobreak >nul

::  3、 start yudao-module-system-biz.jar
start "JAR 3 Window" java -Xms512m -Xmx512m  -Dfile.encoding=UTF-8  -jar "C:\Users\Administrator\Desktop\cloud-jar\yudao-module-system-biz.jar"
:: 等待30秒
timeout /t 30 /nobreak >nul

::  4、start yudao-module-medicine-biz.jar
start "JAR 4 Window" java -Xms512m -Xmx512m -Dfile.encoding=UTF-8  -jar "C:\Users\Administrator\Desktop\cloud-jar\yudao-module-medicine-biz.jar"
echo  all jar start complete！
pause
